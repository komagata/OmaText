cmake_minimum_required(VERSION 3.21)
project(omatext VERSION 0.3.0 LANGUAGES CXX)
# Development tests only. Installing or running the plugin never invokes CMake.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME file-io COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_*.py)
set_tests_properties(file-io PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(ui-test tests/ui_test.cpp tests/support/theme.cpp tests/support/theme.h tests/support/file_io.h)
target_include_directories(ui-test PRIVATE tests/support)
target_link_libraries(ui-test PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Test)
file(GLOB_RECURSE TEST_UI CONFIGURE_DEPENDS tests/support/vendor/qs/*)
qt_add_resources(ui-test test-ui PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${TEST_UI})
target_compile_definitions(ui-test PRIVATE SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME ui COMMAND ui-test)
set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QPA_PLATFORMTHEME=")
